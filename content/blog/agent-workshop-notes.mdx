| Read | The agent can recover the current map. |
| Act | The agent makes the smallest coherent change. |
| Verify | The behavior is proven close to the surface. |

```ts
type WorkState = "read" | "act" | "verify";

export function nextState(state: WorkState) {
  if (state === "read") {
    return "act";
  }

  return "verify";
}
```
